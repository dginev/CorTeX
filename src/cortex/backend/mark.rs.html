19pub(crate) fn mark_imported(
20  connection: &mut PgConnection,
21  imported_tasks: &[NewTask],
22) -> Result<usize, Error> {
23  // Insert, but only if the task is new (allow for extension calls with the same method)
24  insert_into(tasks::table)
25    .values(imported_tasks)
26    .on_conflict_do_nothing()
27    .execute(connection)
28}

30/// **Pause** a `(corpus, service)` run: transition every **in-progress** task (`status >= 0` — i.e.
31/// TODO plus any leased/Queued mark) to **Blocked**, so the ventilator stops handing them out (it
32/// only fetches `status = TODO`). Completed tasks (`status < 0`) are left alone. Returns the number
33/// paused. The exact inverse of [`resume_blocked`] (Arm 7 run-lifecycle control).
34pub(crate) fn mark_blocked(
35  connection: &mut PgConnection,
36  corpus_id_val: i32,
37  service_id_val: i32,
38) -> Result<usize, Error> {
39  update(tasks::table)
40    .filter(tasks::corpus_id.eq(corpus_id_val))
41    .filter(tasks::service_id.eq(service_id_val))
42    .filter(tasks::status.ge(0))
43    .set(tasks::status.eq(TaskStatus::Blocked(-6).raw()))
44    .execute(connection)
45}

47/// **Resume** a paused `(corpus, service)` run: transition every **Blocked** task (`status < -5`)
48/// back to **TODO** (`0`), so the ventilator re-leases it on the next fetch. Returns the number
49/// resumed. The exact inverse of [`mark_blocked`]; completed and already-in-progress tasks are left
50/// alone (Invalid is `-5`, so the `< -5` filter never touches it).
51pub(crate) fn resume_blocked(
52  connection: &mut PgConnection,
53  corpus_id_val: i32,
54  service_id_val: i32,
55) -> Result<usize, Error> {
56  update(tasks::table)
57    .filter(tasks::corpus_id.eq(corpus_id_val))
58    .filter(tasks::service_id.eq(service_id_val))
59    .filter(tasks::status.lt(-5))
60    .set(tasks::status.eq(TaskStatus::TODO.raw()))
61    .execute(connection)
62}

64/// **Pause ALL conversions** (the dashboard's global control): block every in-progress task
65/// (`status >= 0`) across **every** `(corpus, service)`, fleet-wide, so the ventilator stops
66/// leasing new work everywhere. The global twin of [`mark_blocked`]; returns the number paused.
67/// Reversible with [`resume_all_blocked`]. (In-flight tasks already dispatched still land their
68/// results — pause only stops *new* leasing, exactly like the per-run pause.)
69pub(crate) fn mark_all_blocked(connection: &mut PgConnection) -> Result<usize, Error> {
70  update(tasks::table)
71    .filter(tasks::status.ge(0))
72    .set(tasks::status.eq(TaskStatus::Blocked(-6).raw()))
73    .execute(connection)
74}

76/// **Resume ALL conversions**: return every Blocked task (`status < -5`) across every pair to TODO
77/// — the exact inverse of [`mark_all_blocked`]. Returns the number resumed.
78pub(crate) fn resume_all_blocked(connection: &mut PgConnection) -> Result<usize, Error> {
79  update(tasks::table)
80    .filter(tasks::status.lt(-5))
81    .set(tasks::status.eq(TaskStatus::TODO.raw()))
82    .execute(connection)
83}

85pub(crate) fn mark_done(
86  connection: &mut PgConnection,
87  reports: &[TaskReport],
88) -> Result<(), Error> {
89  use crate::schema::tasks::{id, status};
90
91  // One report per task: a batch can carry two for the same task — the reaper's
92  // `never_completed_with_retries` and the late result that replaces it, or two attempts' late
93  // results (KNOWN_ISSUES D-26) — and the batched statements below would apply both (an arbitrary
94  // status, both reports' messages). See `latest_report_per_task` for the one that stands.
95  let reports = latest_report_per_task(reports);
96  // PostgreSQL caps a single statement at 65535 bind parameters. The finalize path
97  // batches across an entire drained burst of reports, so at fleet scale (many papers
98  // × many log messages) the per-severity INSERTs and the `eq_any` id lists overflow
99  // that cap — which made the whole statement fail (`number of parameters must be
100  // between 0 and 65535`), the finalize thread panic, and the dispatcher wedge
101  // (observed on a 64-worker run, 2026-06-17). Chunk every batched statement to stay
102  // under the cap: `eq_any` binds 1 param per id; a log row binds 4 columns
103  // (task_id, category, what, details), so 16k rows ≈ 64k params.
104  const ID_CHUNK: usize = 50_000;
105  const LOG_INSERT_CHUNK: usize = 16_000;
106  connection.transaction::<(), Error, _>(|t_connection| {
107    // A report is applied only under the lease it was dispatched with: `report.task.status`, the
108    // lease mark `fetch_tasks` set (KNOWN_ISSUES D-28). A result applies while the task's row is
109    // still at that mark, or already completed (a negative status: an earlier result of the same
110    // lease, which D-26 lets it replace — or a pause's Blocked); a stand-in (the dead-letter, an
111    // unpublishable archive) only at that mark, so a result persisted first keeps the task. A row
112    // an operator re-ran meanwhile — reset to TODO, or re-leased under a new mark — takes neither:
113    // the old run's result must not count for the new run. (A late result of an old run whose
114    // task the dispatcher still held when it was re-fetched never gets here: the dispatcher
115    // refuses it as `Stale`, since the claim would carry the new lease's mark.) Every report's
116    // row is locked, in one global id order, so a concurrent writer can neither slip in between
117    // nor deadlock with this; `FOR NO KEY UPDATE` does not block the log tables' foreign-key
118    // checks.
119    let mut report_ids: Vec<i64> = reports.iter().map(|report| report.task.id).collect();
120    report_ids.sort_unstable();
121    let mut current: HashMap<i64, i32> = HashMap::with_capacity(report_ids.len());
122    for ids in report_ids.chunks(ID_CHUNK) {
123      let rows: Vec<(i64, i32)> = tasks::table
124        .filter(id.eq_any(ids))
125        .select((id, status))
126        .order(id)
127        .for_no_key_update()
128        .load(t_connection)?;
129      current.extend(rows);
130    }
131    let (reports, dropped): (Vec<&TaskReport>, Vec<&TaskReport>) =
132      reports.into_iter().partition(|report| {
133        current.get(&report.task.id).is_some_and(|&now| {
134          now == report.task.status || (!report.stand_in && now < TaskStatus::TODO.raw())
135        })
136      });
137    if !dropped.is_empty() {
138      let first: Vec<i64> = dropped.iter().take(8).map(|report| report.task.id).collect();
139      tracing::warn!(
140        dropped = dropped.len(),
141        first_task_ids = ?first,
142        "mark_done: reports not applied: their task was re-run (or is gone) since their lease (D-28)"
143      );
144    }
145    // Collect the finalized task ids once, to clear their prior logs in a single batched statement
146    // per table instead of five deletes *per task*. With one report per task, a batched
147    // `task_id = ANY(...)` deletes exactly the same rows as the old per-task loop — far fewer
148    // round-trips on the hot finalize path (KNOWN_ISSUES D-8 write-amplification).
149    let task_ids: Vec<i64> = reports.iter().map(|report| report.task.id).collect();
150    // Clear the prior log messages for every finalized task (one statement per severity
151    // table), chunked over the task-id list to respect the bind-parameter cap.
152    for ids in task_ids.chunks(ID_CHUNK) {
153      delete(log_infos::table.filter(log_infos::task_id.eq_any(ids))).execute(t_connection)?;
154      delete(log_warnings::table.filter(log_warnings::task_id.eq_any(ids)))
155        .execute(t_connection)?;
156      delete(log_errors::table.filter(log_errors::task_id.eq_any(ids))).execute(t_connection)?;
157      delete(log_fatals::table.filter(log_fatals::task_id.eq_any(ids))).execute(t_connection)?;
158      delete(log_invalids::table.filter(log_invalids::task_id.eq_any(ids)))
159        .execute(t_connection)?;
160      delete(task_runtimes::table.filter(task_runtimes::task_id.eq_any(ids)))
161        .execute(t_connection)?;
162    }
163    // Group the finalized task ids by their target status, and partition the new messages by
164    // severity table, in one pass — so both the status UPDATEs and the message INSERTs become a
165    // handful of batched statements below instead of two-per-task in a loop (the finalize hot
166    // path).
167    let mut ids_by_status: HashMap<i32, Vec<i64>> = HashMap::new();
168    let mut new_infos: Vec<NewLogInfo> = Vec::new();
169    let mut new_warnings: Vec<NewLogWarning> = Vec::new();
170    let mut new_errors: Vec<NewLogError> = Vec::new();
171    let mut new_fatals: Vec<NewLogFatal> = Vec::new();
172    let mut new_invalids: Vec<NewLogInvalid> = Vec::new();
173    // Denormalized per-task runtimes, mirrored from each report's `Info:runtime_ms:<N>` line so
174    // the per-service runtime report aggregates over this narrow table instead of re-scanning ~2.8M
175    // `log_infos` rows JOINed to `tasks` on every page view (see migration …_create_task_runtimes).
176    let mut new_runtimes: Vec<NewTaskRuntime> = Vec::new();
177    for report in reports.iter() {
178      ids_by_status
179        .entry(report.status.raw())
180        .or_default()
181        .push(report.task.id);
182      for message in &report.messages {
183        // The synthetic conversion-status message is not a real log entry (kept out of the tables).
184        if message.severity() == "status" {
185          continue;
186        }
187        // The runtime line is also an Info log row; capture its parsed value for `task_runtimes`.
188        // The latexml-oxide worker now emits `Info:runtime_ms:<N>` (category=`runtime_ms`, the
189        // value in `what`) so the runtime surfaces as its own report category + per-value
190        // subreport; older workers emitted `Info:cortex:runtime_ms <N>` (what=`runtime_ms`,
191        // value in `details`). We accept BOTH so a rolling/mixed fleet keeps populating the
192        // per-service runtime report. A malformed value just skips the denormalized row —
193        // the log row is still written below, and the report's other rows are unaffected.
194        if let NewTaskMessage::Info(record) = message {
195          let runtime_value = if record.category == "runtime_ms" {
196            Some(record.what.as_str())
197          } else if record.category == "cortex" && record.what == "runtime_ms" {
198            Some(record.details.as_str())
199          } else {
200            None
201          };
202          if let Some(value) = runtime_value
203            && let Ok(runtime_ms) = value.parse::<i32>()
204          {
205            new_runtimes.push(NewTaskRuntime {
206              task_id: report.task.id,
207              service_id: report.task.service_id,
208              runtime_ms,
209            });
210          }
211        }
212        match message {
213          NewTaskMessage::Info(record) => new_infos.push(record.clone()),
214          NewTaskMessage::Warning(record) => new_warnings.push(record.clone()),
215          NewTaskMessage::Error(record) => new_errors.push(record.clone()),
216          NewTaskMessage::Fatal(record) => new_fatals.push(record.clone()),
217          NewTaskMessage::Invalid(record) => new_invalids.push(record.clone()),
218        }
219      }
220      // TODO: Update dependenct services, when integrated in DB
221    }
222    // Apply the status updates: one batched UPDATE per *distinct* terminal status (a small fixed
223    // set — NoProblem/Warning/Error/Fatal/Invalid), each over the disjoint id set that resolved
224    // to it.
225    for (status_value, status_ids) in &ids_by_status {
226      for ids in status_ids.chunks(ID_CHUNK) {
227        update(tasks::table)
228          .filter(id.eq_any(ids))
229          .set(status.eq(*status_value))
230          .execute(t_connection)?;
231      }
232    }
233    // Batched INSERT per severity table, chunked to respect the bind-parameter cap.
234    // (`chunks` over an empty Vec yields nothing, so no `is_empty` guard is needed.)
235    for chunk in new_infos.chunks(LOG_INSERT_CHUNK) {
236      insert_into(log_infos::table)
237        .values(chunk)
238        .execute(t_connection)?;
239    }
240    for chunk in new_warnings.chunks(LOG_INSERT_CHUNK) {
241      insert_into(log_warnings::table)
242        .values(chunk)
243        .execute(t_connection)?;
244    }
245    for chunk in new_errors.chunks(LOG_INSERT_CHUNK) {
246      insert_into(log_errors::table)
247        .values(chunk)
248        .execute(t_connection)?;
249    }
250    for chunk in new_fatals.chunks(LOG_INSERT_CHUNK) {
251      insert_into(log_fatals::table)
252        .values(chunk)
253        .execute(t_connection)?;
254    }
255    for chunk in new_invalids.chunks(LOG_INSERT_CHUNK) {
256      insert_into(log_invalids::table)
257        .values(chunk)
258        .execute(t_connection)?;
259    }
260    // Denormalized runtimes: the prior runtime rows for these tasks were cleared in the delete loop
261    // above, so a plain insert is correct (3 columns/row → 16k rows ≈ 48k binds, under the cap).
262    for chunk in new_runtimes.chunks(LOG_INSERT_CHUNK) {
263      insert_into(task_runtimes::table)
264        .values(chunk)
265        .execute(t_connection)?;
266    }
267    Ok(())
268  })?;
269  Ok(())
270}

272/// One report for each task in `reports`, in the order of the kept reports: the last *result*
273/// for the task, or, when it has none, its last stand-in ([`TaskReport::stand_in`], e.g. the
274/// reaper's dead-letter) — the
275/// reaper's `never_completed` stand-in never replaces an attempt's result, whichever reached the
276/// finalize channel first (KNOWN_ISSUES D-26).
277pub(crate) fn latest_report_per_task(reports: &[TaskReport]) -> Vec<&TaskReport> {
278  let mut last: HashMap<i64, usize> = HashMap::with_capacity(reports.len());
279  for (index, report) in reports.iter().enumerate() {
280    let replaces = match last.get(&report.task.id) {
281      Some(&kept) => !report.stand_in || reports[kept].stand_in,
282      None => true,
283    };
284    if replaces {
285      last.insert(report.task.id, index);
286    }
287  }
288  reports
289    .iter()
290    .enumerate()
291    .filter(|(index, report)| last.get(&report.task.id) == Some(index))
292    .map(|(_, report)| report)
293    .collect()
294}

296pub(crate) fn mark_rerun<'a>(
297  connection: &'a mut PgConnection,
298  options: RerunOptions<'a>,
299) -> Result<(), Error> {
300  let RerunOptions {
301    corpus,
302    service,
303    severity_opt,
304    category_opt,
305    what_opt,
306    owner_opt,
307    description_opt,
308  } = options;
309  use crate::schema::tasks::{corpus_id, service_id, status};
310  // We are starting a new run, first catalog the current metadata in our historical records.
311  let mut description = description_opt.unwrap_or_else(|| String::from("mark for rerun "));
312  // auto-generate a report message from the selected filters
313  description.push_str("(filters:");
314  if severity_opt.is_none() && category_opt.is_none() && what_opt.is_none() {
315    description.push_str(" entire corpus");
316  } else {
317    if let Some(ref severity) = severity_opt {
318      description.push_str(" severity=");
319      description.push_str(severity);
320    }
321    if let Some(ref category) = category_opt {
322      description.push_str(" category=");
323      description.push_str(category);
324    }
325    if let Some(ref what) = what_opt {
326      description.push_str(" what=");
327      description.push_str(what);
328    }
329  }
330  description.push(')');
331
332  // Atomic rerun (R-11): the run record + the two-phase task reset (set the scope to a temporary
333  // `mark`, delete its logs, flip `mark → TODO`) must all-or-nothing together. Otherwise a frontend
334  // crash/restart mid-rerun could strand the whole scope in the `mark` value — a positive status
335  // the dispatcher won't lease (it leases `TODO=0`) — so the tasks never re-convert until a
336  // dispatcher restart's limbo recovery. One transaction makes the temporary `mark` never
337  // observable and a crash a clean rollback. `mark_new_run`'s own transaction nests here as a
338  // savepoint. (The closure param shadows `connection` so the body below uses the transaction
339  // connection.)
340  connection.transaction::<(), Error, _>(|connection| {
341    mark_new_run(
342      connection,
343      corpus,
344      service,
345      owner_opt.unwrap_or_else(|| "admin".to_string()),
346      description,
347    )?;
348    // Rerun = set status to TODO for all tasks, deleting old logs. The temporary `mark` is a
349    // positive sentinel drawn *above* the max lease (R-13): we re-select the scope below by
350    // `status = mark`, so it must not collide with a live in-flight lease (`[1, 65536]`) or `TODO`
351    // (0) — else an unrelated task would be swept into the rerun and double-dispatched.
352    let mark: i32 = rerun_mark();
353
354    // First, mark as blocked all of the tasks in the chosen scope, using a special mark
355    match severity_opt {
356      Some(severity) => match category_opt {
357        Some(category) => match what_opt {
358          // All tasks in a "what" class
359          Some(what) => match severity.to_lowercase().as_str() {
360            "warning" => LogWarning::mark_rerun_by_what(
361              mark, corpus.id, service.id, &category, &what, connection,
362            ),
363            "error" => LogError::mark_rerun_by_what(
364              mark, corpus.id, service.id, &category, &what, connection,
365            ),
366            "fatal" => LogFatal::mark_rerun_by_what(
367              mark, corpus.id, service.id, &category, &what, connection,
368            ),
369            "invalid" => LogInvalid::mark_rerun_by_what(
370              mark, corpus.id, service.id, &category, &what, connection,
371            ),
372            _ => {
373              LogInfo::mark_rerun_by_what(mark, corpus.id, service.id, &category, &what, connection)
374            },
375          }?,
376          // None: All tasks in a category
377          None => match severity.to_lowercase().as_str() {
378            "warning" => {
379              LogWarning::mark_rerun_by_category(mark, corpus.id, service.id, &category, connection)
380            },
381            "error" => {
382              LogError::mark_rerun_by_category(mark, corpus.id, service.id, &category, connection)
383            },
384            "fatal" => {
385              LogFatal::mark_rerun_by_category(mark, corpus.id, service.id, &category, connection)
386            },
387            "invalid" => {
388              LogInvalid::mark_rerun_by_category(mark, corpus.id, service.id, &category, connection)
389            },
390            _ => {
391              LogInfo::mark_rerun_by_category(mark, corpus.id, service.id, &category, connection)
392            },
393          }?,
394        },
395        None => {
396          // All tasks in a certain status/severity
397          let status_to_rerun: i32 = TaskStatus::from_key(&severity)
398            .unwrap_or(TaskStatus::NoProblem)
399            .raw();
400          update(tasks::table)
401            .filter(corpus_id.eq(corpus.id))
402            .filter(service_id.eq(service.id))
403            .filter(status.eq(status_to_rerun))
404            .set(status.eq(mark))
405            .execute(connection)?
406        },
407      },
408      None => {
409        // Entire corpus
410        update(tasks::table)
411          .filter(corpus_id.eq(corpus.id))
412          .filter(service_id.eq(service.id))
413          .filter(status.lt(0))
414          .set(status.eq(mark))
415          .execute(connection)?
416      },
417    };
418
419    // Next, delete all logs for the blocked tasks.
420    // Note that if we are using a negative blocking status, this query should get sped up via an
421    // "Index Scan using log_taskid on logs"
422    let affected_tasks = tasks::table
423      .filter(corpus_id.eq(corpus.id))
424      .filter(service_id.eq(service.id))
425      .filter(status.eq(mark));
426    let affected_tasks_ids = affected_tasks.select(tasks::id);
427
428    let affected_log_infos = log_infos::table.filter(log_infos::task_id.eq_any(affected_tasks_ids));
429    delete(affected_log_infos).execute(connection)?;
430    let affected_log_warnings =
431      log_warnings::table.filter(log_warnings::task_id.eq_any(affected_tasks_ids));
432    delete(affected_log_warnings).execute(connection)?;
433    let affected_log_errors =
434      log_errors::table.filter(log_errors::task_id.eq_any(affected_tasks_ids));
435    delete(affected_log_errors).execute(connection)?;
436    let affected_log_fatals =
437      log_fatals::table.filter(log_fatals::task_id.eq_any(affected_tasks_ids));
438    delete(affected_log_fatals).execute(connection)?;
439    let affected_log_invalids =
440      log_invalids::table.filter(log_invalids::task_id.eq_any(affected_tasks_ids));
441    delete(affected_log_invalids).execute(connection)?;
442
443    // Lastly, switch all blocked tasks to TODO, and complete the rerun mark pass.
444    update(affected_tasks)
445      .set(status.eq(TaskStatus::TODO.raw()))
446      .execute(connection)?;
447
448    // The reran scope's reports are now stale (logs deleted, statuses reset). Drop its cached
449    // report grains inside the same transaction so the next report view repopulates from the
450    // fresh data — scoped to exactly this (corpus, service), never the global cube.
451    super::rollup::invalidate_scope(connection, corpus.id, service.id)?;
452
453    Ok(())
454  })
455}

457pub(crate) fn mark_new_run(
458  connection: &mut PgConnection,
459  corpus: &Corpus,
460  service: &Service,
461  owner: String,
462  description: String,
463) -> Result<(), Error> {
464  // Step 1. Mark any open runs as completed.
465  mark_run_completed(connection, corpus, service)?;
466  // Step 2. Create this historical run
467  let hrun = NewHistoricalRun {
468    corpus_id: corpus.id,
469    service_id: service.id,
470    description,
471    owner,
472  };
473  hrun.create(connection)?;
474  // NB: this used to synchronously `REFRESH MATERIALIZED VIEW report_summary` here so the run
475  // boundary showed up in reports immediately — but that is a ~2 min rebuild at production scale,
476  // and `mark_new_run` runs on the rerun *request* thread, so it blocked the HTTP response for
477  // minutes (KNOWN_ISSUES R-5). The refresh is now spawned **off the request path** by the rerun
478  // entry points (`reports::rerun_report`, `concerns::serve_rerun`) via
479  // `jobs::spawn_report_refresh`, and the dispatcher refreshes on drain + the regular interval.
480  // Bookkeeping no longer triggers a refresh.
481  Ok(())
482}

484fn mark_run_completed(
485  connection: &mut PgConnection,
486  corpus: &Corpus,
487  service: &Service,
488) -> Result<(), Error> {
489  let to_finish: Vec<HistoricalRun> = HistoricalRun::find_by(corpus, service, connection)?
490    .into_iter()
491    .filter(|run| run.end_time.is_none())
492    .collect();
493  if !to_finish.is_empty() {
494    connection.transaction::<(), Error, _>(move |t_connection| {
495      for run in to_finish.into_iter() {
496        run.mark_completed(t_connection)?;
497      }
498      Ok(())
499    })?;
500  }
501  Ok(())
502}

512/// Freeze the current per-task statuses of a `(corpus, service)` into `historical_tasks` — the
513/// id-keyed core of [`save_historical_tasks`]. Called both by the human "save snapshot" and, on
514/// **run-completion-on-drain**, to capture the just-finished run's outcomes as the **baseline** for
515/// the next run's live run-diff (see `backend::Backend::complete_run_if_drained`). One
516/// `INSERT … SELECT` of scope-size rows; retention/pruning of stale snapshots is a follow-up.
517pub fn snapshot_tasks(
518  connection: &mut PgConnection,
519  corpus_id: i32,
520  service_id: i32,
521) -> Result<usize, Error> {
522  insert_into(historical_tasks::table)
523    .values(
524      tasks::table
525        .select((tasks::id, tasks::status))
526        .filter(tasks::corpus_id.eq(corpus_id))
527        .filter(tasks::service_id.eq(service_id)),
528    )
529    .into_columns((historical_tasks::task_id, historical_tasks::status))
530    .execute(connection)
531}

533#[cfg(test)]
534mod tests {
535  use super::latest_report_per_task;
536  use crate::helpers::{NEVER_COMPLETED_WHAT, NewTaskMessage, TaskReport, TaskStatus};
537  use crate::models::Task;
538
539  fn report(id: i64, status: TaskStatus) -> TaskReport {
540    TaskReport {
541      stand_in: false,
542      task: Task {
543        id,
544        service_id: 3,
545        corpus_id: 1,
546        status: 7,
547        entry: String::new(),
548      },
549      status,
550      messages: Vec::new(),
551    }
552  }
553
554  #[test]
555  fn the_later_report_for_a_task_stands() {
556    // The reaper's never_completed Fatal, then the late result that replaces it, in one batch.
557    let batch = [
558      report(1, TaskStatus::Fatal),
559      report(2, TaskStatus::Warning),
560      report(1, TaskStatus::NoProblem),
561    ];
562    let kept: Vec<(i64, i32)> = latest_report_per_task(&batch)
563      .iter()
564      .map(|r| (r.task.id, r.status.raw()))
565      .collect();
566    assert_eq!(
567      kept,
568      vec![
569        (2, TaskStatus::Warning.raw()),
570        (1, TaskStatus::NoProblem.raw())
571      ]
572    );
573  }
574
575  fn dead_letter(id: i64) -> TaskReport {
576    TaskReport {
577      messages: vec![NewTaskMessage::new(
578        id,
579        "fatal",
580        "cortex".to_string(),
581        NEVER_COMPLETED_WHAT.to_string(),
582        String::new(),
583      )],
584      stand_in: true,
585      ..report(id, TaskStatus::Fatal)
586    }
587  }
588
589  #[test]
590  fn a_dead_letter_never_replaces_a_result() {
591    // The late result reached the finalize channel before the reaper's dead-letter (a backpressure
592    // race): the result still stands.
593    let batch = [report(1, TaskStatus::Error), dead_letter(1), dead_letter(2)];
594    assert!(batch[1].stand_in && !batch[0].stand_in);
595    let kept: Vec<(i64, bool)> = latest_report_per_task(&batch)
596      .iter()
597      .map(|r| (r.task.id, r.stand_in))
598      .collect();
599    assert_eq!(kept, vec![(1, false), (2, true)]);
600  }
